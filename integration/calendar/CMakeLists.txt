cmake_minimum_required(VERSION 3.20)
project(BoringCalendarFixture LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_path(CALENDAR_INCLUDE_DIR CalendarEvents/CalendarEventsPlugin PATH_SUFFIXES KF6/KDeclarative REQUIRED)
find_library(CALENDAR_LIBRARY NAMES KF6CalendarEvents libKF6CalendarEvents.so.6 REQUIRED)
add_library(pimevents MODULE fixture.cpp)
target_include_directories(pimevents PRIVATE "${CALENDAR_INCLUDE_DIR}")
target_link_libraries(pimevents PRIVATE Qt6::Core "${CALENDAR_LIBRARY}")
set_target_properties(pimevents PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/plasmacalendarplugins")
